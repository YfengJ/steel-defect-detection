- [ ] **Step 4: Implement minimal validators**

```python
from pathlib import Path
from typing import TypeAlias

import yaml

PathLike: TypeAlias = str | Path


def _path(value: PathLike) -> Path:
    return Path(value).expanduser()


def validate_model_path(value: PathLike) -> str | None:
    path = _path(value)
    if not path.is_file():
        return f"找不到模型权重：{path}。请先在本地准备 .pt 权重文件。"
    return None
```

Implement these exact public functions:

```python
def validate_file_source(value: PathLike, label: str = "输入文件") -> str | None:
    path = _path(value)
    return None if path.is_file() else f"找不到{label}：{path}。请重新选择本地文件。"


def validate_directory_source(value: PathLike, label: str = "输入目录") -> str | None:
    path = _path(value)
    return None if path.is_dir() else f"找不到{label}：{path}。请重新选择本地目录。"


def validate_video_source(value: PathLike) -> str | None:
    text = str(value).strip()
    if text.isdecimal():
        return None
    return validate_file_source(text, "视频源")
```

`validate_dataset_config(value)` must load YAML with `yaml.safe_load`, resolve a
relative `path` from the YAML file's parent directory, require local `train` and
`val` entries, check each image path, derive the corresponding label path by
replacing the `images` path component with `labels`, and return a Chinese error
for a missing config, malformed YAML, missing split, missing image path, or
missing label path.
